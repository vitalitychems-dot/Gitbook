+# Quiet gate timer (Gitbook #72 idea 5), rev 2
+
+**Change after Moderator FAIL @6e9b0e49:** the default no longer backs off. Each 5-min tick does a conditional GET with `If-None-Match`. A 304 is free (it doesn't count against the GitHub rate limit) and means IDLE; a 200 with a new ETag means RUN the full gate check. Backoff is now opt-in (`QB_CAP>5`) and is not recommended, because every backoff setting regressed latency (table).
+
+**Reproduce from this PR:** `python3 replay.py` (real data in `data/`):
+- `pushes.txt`: 601 TESSERA commit times, 2026-10-06 → 10-13 (7 d 8 h);
+
+Latency is mean push-detect latency, bootstrap 95% CI over pushes, 5 seeded runs (all 5 within ±0.01 of this one).

+#!/usr/bin/env python3
+"""Quiet gate timer (Gitbook #72 idea 5). Token-free.
+Default (ETag mode): cron ticks every BASE min and does a conditional GET of the head ref with If-None-Match.
+  304 -> IDLE (free: 304s don't count against the GitHub rate limit); 200 with a new ETag/SHA -> RUN the full gate check.
+  Detect latency = the fixed-cron baseline; full checks only happen on change.
+Optional backoff: QB_CAP>BASE doubles the tick interval while idle (fewer wakes, but higher latency; see README).
+CLI: quiet_backoff.py due STATE            -> exit 0 if this tick should poll, 1 to skip (no API call)
+     quiet_backoff.py seen STATE STATUS ETAG -> prints RUN or IDLE and updates STATE"""
+import json,sys,time,os
+BASE=int(os.environ.get("QB_BASE",5));CAP=int(os.environ.get("QB_CAP",BASE))
+def due(st,now): return now>=st.get("next",0)
+def seen(st,status,etag,now,base=BASE,cap=CAP):
+    if int(status)==200 and st.get("etag")!=etag: return "RUN",{"etag":etag,"iv":base,"next":now+base*60}
+    iv=min(st.get("iv",base)*2,cap); return "IDLE",{"etag":st.get("etag"),"iv":iv,"next":now+iv*60}
+if __name__=="__main__":
+    cmd,f=sys.argv[1],sys.argv[2];now=time.time()
+    st=json.load(open(f)) if os.path.exists(f) else {}
+    if cmd=="due": sys.exit(0 if due(st,now) else 1)
+    a,st=seen(st,sys.argv[3],sys.argv[4],now);json.dump(st,open(f,"w"));print(a)

+"""Replay real TESSERA commit times (data/pushes.txt) through cron policies. Reproduces PR numbers.
+Usage: python3 replay.py."""
+import random,statistics as S,os
+from datetime import datetime as D,timedelta as TD
+H=os.path.dirname(os.path.abspath(__file__))
+def load(n): return open(f"{H}/data/{n}").read().splitlines()

+def sim(base=5,cap=5,quiet_h=0,etag=False):
+    """returns (wakes/day, full checks/day, latencies). quiet_h: back off only after quiet_h hours with no push.
+    etag: every base-min tick does a conditional GET (304 free, not counted); full check only on change."""
+    s,e=P[0],P[-1];now=s;iv=base;pi=0;last=s;wakes=full=0;lat=[]
+    while now<e:
+        now+=TD(minutes=iv);wakes+=1;new=[p for p in P[pi:] if p<=now]
+        if etag: full+=bool(new)
+        else: full+=1
+        if new: lat+=[(now-p).total_seconds()/60 for p in new];pi+=len(new);last=new[-1];iv=base
+        elif (now-last).total_seconds()>=quiet_h*3600: iv=min(iv*2,cap)
+    d=(e-s).total_seconds()/86400;return wakes/d,full/d,lat
+def ci(x,seed):
+    r=random.Random(seed);b=sorted(S.mean(r.choices(x,k=len(x))) for _ in range(5000));return S.mean(x),b[125],b[4874]
