If you prefer to run it all from a .py file (inside Python instead of the terminal), here’s a simple version that automates the same setup logic from within Python itself:

import os, tarfile, subprocess, sys
# 1. Extract archive
if not os.path.exists("chimera_complete_final"):
    with tarfile.open("chimera_complete_final.tar.gz", "r:gz") as tar:
        tar.extractall()
# 2. Move into project directory
os.chdir("chimera_complete_final/project_chimera")
# 3. Create virtual environment
subprocess.run([sys.executable, "-m", "venv", "venv"])
# 4. Install dependencies
req = "requirements.txt"
pip_exec = "venv/bin/pip" if os.name != "nt" else "venv\\Scripts\\pip.exe"
if os.path.exists(req):
    subprocess.run([pip_exec, "install", "-r", req])
else:
    subprocess.run([pip_exec, "install", "numpy", "torch", "transformers", "flask"])
# 5. Run the main file
main_files = ["chimera_ultimate.py", "demo.py", "core/chimera_consciousness.py"]
for f in main_files:
    if os.path.exists(f):
        subprocess.run(["venv/bin/python" if os.name != "nt" else "venv\\Scripts\\python.exe", f])
        break
else:
    print("❌ No Chimera main file found.")
